```python
# a...b...c where [a,b) is sorted and [b,c) is sorted
def inplace_merge(a, b, c):
    # a...b...c
    # where [a...b) and [b....c) are sorted
    res = []
    i1, i2 = a, b
    while i1 < b and i2 < c:
        if nums[i1] < nums[i2]:
            res.append(nums[i1])
            i1 += 1
        else:
            res.append(nums[i2])
            i2 += 1
    res += nums[i1:b]
    res += nums[i2:c]
    nums[a:c] = res
```
